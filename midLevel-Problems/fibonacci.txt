
import Foundation


func fibonacciNumbers(_ n:Int){
    
    if n<1{
        print("Invalid Number")
        return
    }
    
    var firstNumber  = 0
    var secondNumber = 1
    for i in 1...n{
        if i == 1{
            print(firstNumber, terminator: " ")
            continue
        }
        print(secondNumber, terminator: " ")
        let thirdNumber  = firstNumber + secondNumber
        firstNumber = secondNumber
        secondNumber = thirdNumber
    }
    
}


print("Enter the value of n")
var n:Int = Int(readLine() ?? "0")!
print(fibonacciNumbers(n))
