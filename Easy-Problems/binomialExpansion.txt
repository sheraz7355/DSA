import Foundation


func binomialCoefficient(_ n: Int, _ r: Int) -> Int {
    let Factorial = { (n:Int)->Int in
        var sum = 1
        for i in 1...n {
            sum *= i
        }
        return sum
    }
   
    return Factorial(n) / (Factorial(r) * Factorial(n-r))
}

print("Enter the value of n")
var n:Int = Int(readLine() ?? "0")!
print("Enter the value of r")
var r:Int = Int(readLine() ?? "0")!
print(binomialCoefficient(n, r))
