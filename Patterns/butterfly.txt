
import Foundation

//  Butterfly n = 4
//   *       *
//   **     **
//   ***   ***
//   *********
//   ***   ***
//   **     **
//   *       *

func hollowDiamond(_ n: Int) {
    let mid = (2 * n - 1) / 2

    for i in 0..<2 * n - 1 {
       
        for j in 0..<2*n{
            
            if (j <= i || j >= (mid+mid+1)-i) && i <= mid {
                print("*",terminator: "")
            } else if i>mid && ( j <= (mid+mid)-i || j >= i+1){
                print("*",terminator: "")
            }else{
                print( " ",terminator: "")
            }
        }
        print()
    }
}




print("Enter The Number")

guard let n = Int(readLine() ?? "0") else {
    print("Invalid Number")
    exit(0)
}

hollowDiamond(n)
